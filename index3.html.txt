  <script>
    // HỆ THỐNG ĐIỀU KHIỂN SƯ PHẠM VÀ LOGIC BÀI HỌC
    AFRAME.registerComponent('edu-system', {
      init: function () {
        this.currentLesson = 1; // Bài 1: Bật/Tắt | Bài 2: Cổng kết nối
        this.caseOn = false;
        this.monitorOn = false;
        this.isShutdown = false;
        this.menuOpen = false;
        this.score = 0;

        // Truy xuất phần tử DOM 3D
        this.caseBtn = document.querySelector('#case-btn');
        this.caseLed = document.querySelector('#case-led');
        this.monitorBtn = document.querySelector('#monitor-btn');
        this.monitorScreen = document.querySelector('#monitor-screen');
        this.desktopUI = document.querySelector('#desktop-ui');
        this.winStartBtn = document.querySelector('#win-start-btn');
        this.winPowerBtn = document.querySelector('#win-power-btn');
        this.winMenu = document.querySelector('#win-start-menu');

        // Âm thanh
        this.sndClick = document.querySelector('#snd-click');
        this.sndFan = document.querySelector('#snd-fan');
        this.sndWin = document.querySelector('#snd-win');

        // 1. Thao tác nút nguồn Case
        this.caseBtn.addEventListener('click', () => {
          this.sndClick.components.sound.playSound();
          if (!this.caseOn) {
            this.caseOn = true;
            this.caseLed.setAttribute('material', 'color', '#00E676');
            this.sndFan.components.sound.playSound();
            this.addScore(25);
            this.updateDisplay();
            this.setGuide("✅ Thùng máy đã khởi động! Bước tiếp theo: Em hãy bấm nút nguồn trên Màn hình.");
          } else {
            if (!this.isShutdown) {
              this.setGuide("⚠️ CẢNH BÁO: Tắt đột ngột bằng nút Case sẽ làm hỏng ổ đĩa và mất bài tập! Hãy tắt từ màn hình trước.", "#FF5252");
            } else {
              this.caseOn = false;
              this.caseLed.setAttribute('material', 'color', '#333333');
              this.sndFan.components.sound.stopSound();
              this.addScore(25);
              this.updateDisplay();
              this.setGuide("🎉 HOÀN THÀNH QUY TRÌNH! Em là một học sinh sử dụng máy tính rất an toàn và đúng cách!", "#69F0AE");
            }
          }
        });

        // 2. Thao tác nút Màn hình
        this.monitorBtn.addEventListener('click', () => {
          this.sndClick.components.sound.playSound();
          this.monitorOn = !this.monitorOn;
          if (this.monitorOn && this.caseOn) this.addScore(25);
          this.updateDisplay();
        });

        // 3. Nhấp biểu tượng Windows Start
        this.winStartBtn.addEventListener('click', () => {
          this.sndClick.components.sound.playSound();
          this.menuOpen = !this.menuOpen;
          this.winMenu.setAttribute('visible', this.menuOpen);
        });

        // 4. Nhấp nút Shut Down trên Windows
        this.winPowerBtn.addEventListener('click', () => {
          this.sndClick.components.sound.playSound();
          this.isShutdown = true;
          this.winMenu.setAttribute('visible', 'false');
          this.winStartBtn.setAttribute('visible', 'false');
          this.addScore(25);
          
          this.setGuide("🔄 Windows đang lưu dữ liệu và tắt an toàn... Em chờ đèn thùng máy tắt rồi tắt màn hình nhé!", "#FFD54F");

          setTimeout(() => {
            this.caseOn = false;
            this.caseLed.setAttribute('material', 'color', '#333333');
            this.sndFan.components.sound.stopSound();
            this.updateDisplay();
            this.setGuide("Thùng máy đã tắt! Cuối cùng, hãy bấm tắt nút nguồn màn hình để tiết kiệm điện.");
          }, 3000);
        });

        // 5. Tương tác Bài 2: Cổng kết nối phía sau
        document.querySelectorAll('.port-item').forEach(port => {
          port.addEventListener('click', () => {
            this.sndClick.components.sound.playSound();
            const name = port.getAttribute('data-name');
            const desc = port.getAttribute('data-desc');
            this.setGuide(`🔍 ${name}: ${desc}`, "#80D8FF");
          });
        });
      },

      updateDisplay: function () {
        if (this.caseOn && this.monitorOn) {
          this.monitorScreen.setAttribute('material', 'color', '#1565C0');
          this.desktopUI.setAttribute('visible', 'true');
          this.winStartBtn.setAttribute('visible', 'true');
          if (this.score < 50) this.setGuide("Em đã mở máy tính thành công! Hãy quan sát màn hình nền Windows.");
        } else if (!this.caseOn && this.monitorOn) {
          this.monitorScreen.setAttribute('material', 'color', '#263238');
          this.desktopUI.setAttribute('visible', 'false');
          this.winMenu.setAttribute('visible', 'false');
          this.setGuide("Màn hình đã bật nhưng Thùng máy chưa bật (Báo No Signal). Em hãy bấm nút trên thùng máy!");
        } else {
          this.monitorScreen.setAttribute('material', 'color', '#0a0a0a');
          this.desktopUI.setAttribute('visible', 'false');
          this.winMenu.setAttribute('visible', 'false');
        }
      },

      addScore: function (pts) {
        this.score = Math.min(100, this.score + pts);
        document.querySelector('#score-badge').innerText = `Điểm thực hành: ${this.score}/100`;
        if (this.score === 100) {
          this.sndWin.components.sound.playSound();
        }
      },

      setGuide: function (msg, color = '#FFF9C4') {
        const txt = document.querySelector('#guide-text');
        txt.innerText = msg;
        txt.style.color = color;
      }
    });

    // CHUYỂN GÓC NHÌN CAMERA VÀ BÀI HỌC
    function switchView(mode) {
      const rig = document.querySelector('#camera-rig');
      document.querySelectorAll('.hud-btn').forEach(b => b.classList.remove('active'));

      if (mode === 'overview') {
        rig.setAttribute('animation', 'property: position; to: 0 1.25 0; dur: 800; easing: easeInOutQuad');
        document.querySelector('#btn-view-all').classList.add('active');
        document.querySelector('#lesson-title').innerText = "BÀI HỌC 1: QUY TRÌNH BẬT TẮT MÁY TÍNH";
      } else if (mode === 'screen') {
        rig.setAttribute('animation', 'property: position; to: 0.1 1.05 -0.92; dur: 800; easing: easeInOutQuad');
        document.querySelector('#btn-view-screen').classList.add('active');
      } else if (mode === 'case') {
        rig.setAttribute('animation', 'property: position; to: -0.45 1.05 -0.95; dur: 800; easing: easeInOutQuad');
        document.querySelector('#btn-view-case').classList.add('active');
      } else if (mode === 'back') {
        // Góc nhìn mặt sau thùng máy (Bài học 2)
        rig.setAttribute('animation', 'property: position; to: -0.45 1.05 -2.1; dur: 800; easing: easeInOutQuad');
        rig.setAttribute('animation__rot', 'property: rotation; to: 0 180 0; dur: 800');
        document.querySelector('#btn-view-back').classList.add('active');
        document.querySelector('#lesson-title').innerText = "BÀI HỌC 2: KHÁM PHÁ CÁC CỔNG KẾT NỐI PHÍA SAU";
        document.querySelector('#guide-text').innerText = "👉 Em hãy nhấp chuột vào các cổng cắm phía sau thùng máy để tìm hiểu công dụng!";
      }
    }
  </script>
